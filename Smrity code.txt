9. Program to Find Factorial Using Function

#include <stdio.h>

int factorial(int n){
    int i, fact = 1;

    for(i = 1; i <= n; i++){
        fact *= i;
    }

    return fact;
}

int main(){
    printf("Factorial = %d", factorial(5));
    return 0;
}


10. Program to Find Largest Number Among Three Numbers

#include <stdio.h>

int largest(int a, int b, int c){
    if(a > b && a > c)
        return a;
    else if(b > c)
        return b;
    else
        return c;
}

int main(){
    printf("Largest = %d", largest(5, 9, 3));
    return 0;
}

11. Recursive Factorial Program

#include <stdio.h>

int factorial(int n){
    if(n == 0)
        return 1;
    else
        return n * factorial(n - 1);
}

int main(){
    printf("Factorial = %d", factorial(5));
    return 0;
}

12. Program to Swap Two Numbers Using Function

#include <stdio.h>

void swap(int *a, int *b){
    int temp = *a;
    *a = *b;
    *b = temp;
}

int main(){
    int x = 5, y = 10;

    swap(&x, &y);

    printf("x = %d, y = %d", x, y);

    return 0;
}

13. Recursive Fibonacci Series Program

#include <stdio.h>

int fib(int n){
    if(n == 0)
        return 0;

    if(n == 1)
        return 1;

    return fib(n - 1) + fib(n - 2);
}

int main(){
    printf("%d", fib(6));
    return 0;
}

14. Program to Find Largest Number in an Array

#include <stdio.h>

int main(){
    int num[5] = {12, 4, 56, 7, 24};
    int i, max;

    max = num[0];

    for(i = 1; i < 5; i++){
        if(num[i] > max){
            max = num[i];
        }
    }

    printf("Largest = %d", max);

    return 0;
}
